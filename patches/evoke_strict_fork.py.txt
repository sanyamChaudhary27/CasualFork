EVOKE_STRICT_FORK_JSON      fork-boundary hook config; inline JSON or path:

def run_id():
    """Stable per-process run identity (env-declared wins)."""
    if "run" in _RUN_CACHE:
        return _RUN_CACHE["run"]
    rid = os.environ.get(ENV_RUN_ID)
    if not rid:
        import uuid
        rid = "run-" + uuid.uuid4().hex[:12]
    _RUN_CACHE["run"] = rid
    return rid


def pair_id():
    if "pair" in _RUN_CACHE:
        return _RUN_CACHE["pair"]
    pid = os.environ.get(ENV_PAIR_ID) or "pair-unset"
    _RUN_CACHE["pair"] = pid
    return pid


def new_render_call():
    """Monotonic per-render-call id for R7 isolated-stream witnessing."""
    with _LOCK:
        _RENDER_CALL["n"] += 1
        return _RENDER_CALL["n"]


def ledger_enabled():
    return bool(os.environ.get(ENV_LEDGER))


def fork_cfg():
    """Parse EVOKE_STRICT_FORK_JSON (inline JSON object, or a path to one)."""
    global _FORK_CFG, _FORK_CFG_RAW
    raw = os.environ.get(ENV_FORK)
    if raw == _FORK_CFG_RAW:
        return _FORK_CFG
    cfg = None
    if raw:
        txt = raw
        if not raw.lstrip().startswith("{"):
            with open(raw, "r", encoding="utf-8") as fh:
                txt = fh.read()
        cfg = json.loads(txt)
    _FORK_CFG = cfg
    _FORK_CFG_RAW = raw
    return _FORK_CFG


def active():
    return ledger_enabled() or fork_cfg() is not None


def sha256_bytes(data):
    return hashlib.sha256(data).hexdigest()


def _byte_payload(t):
    tb = t.detach().contiguous().cpu()
    shape_s, dtype_s = str(list(tb.shape)), str(tb.dtype)
    try:
        raw = tb.view(torch.uint8).reshape(-1).numpy().tobytes()
    except Exception:
        raw = tb.reshape(-1).to(torch.float64).numpy().tobytes()
    return raw, shape_s, dtype_s


def tensor_sha256(t):
    if not torch.is_tensor(t):
        return None
    raw, shape_s, dtype_s = _byte_payload(t)
    h = hashlib.sha256()
    h.update(shape_s.encode()); h.update(b"|"); h.update(dtype_s.encode()); h.update(b"|")
    h.update(raw)
    return h.hexdigest()


def gen_state_of(generator):
    """sha256 of the generator state; None when inactive or generator None."""
    if not active():
        return None
    g = generator[0] if isinstance(generator, list) else generator
    if g is None:
        return None
    st = g.get_state()
    return sha256_bytes(st.contiguous().reshape(-1).numpy().tobytes())


def global_rng_digest():
    if not active():
        return None
    try:
        st = torch.get_rng_state()
        return sha256_bytes(st.contiguous().reshape(-1).numpy().tobytes())
    except Exception:
        return None


def pixel_diag(pix, mean, std):
    """R2/R3 diagnostics: byte-equality witnesses for encode inputs/params."""
    if not ledger_enabled():
        return None
    try:
        return {
            "sha256_input_pixels": tensor_sha256(pix),
            "sha256_mean": tensor_sha256(mean),
            "sha256_std": tensor_sha256(std),
        }
    except Exception:
        return {}


class NoiseTape:
    """Ordered record of instrumented draws; backs the JSONL ledger."""

    def __init__(self, branch=None):
        self.branch = branch or get_branch_id()
        self.entries = []

    def record(self, site_key, tensor, **meta):
        entry = dict(meta)
        entry.setdefault("event", "draw")
        entry["site_id"] = site_key
        if tensor is None:
            entry["shape"] = None
            entry["dtype"] = None
            entry["tensor_sha256"] = None
        else:
            entry["shape"] = list(tensor.shape) if torch.is_tensor(tensor) else None
            entry["dtype"] = str(tensor.dtype) if torch.is_tensor(tensor) else None
            entry["tensor_sha256"] = tensor_sha256(tensor)
        self.entries.append(entry)
        return entry

    def replay(self, site_key):
        """PHASE-2 FALLBACK: not implemented in evoke-74d26851-strict-coupling."""
        raise NotImplementedError(
            "NoiseTape.replay is a PHASE-2 FALLBACK (planned after G1); "
            "this patch records draws but provides no replay path."
        )

    def to_jsonl(self, path):
        with open(path, "w", encoding="utf-8", newline="\n") as fh:
            for entry in self.entries:
                fh.write(json.dumps(entry, sort_keys=True) + "\n")


_TAPE = NoiseTape()
_SEQ = {"n": 0}
_SEQ_LOCK = threading.Lock()


def _next_seq():
    """One monotonic line counter shared by draws AND events, so event
    ordering against draws (e.g., restore-before-first-post-fork-draw) is
    expressible in the ledger itself. Uses its OWN lock: callers invoke this
    while already holding _LOCK (a non-reentrant Lock), so acquiring _LOCK
    here would deadlock."""
    with _SEQ_LOCK:
        _SEQ["n"] += 1
        return _SEQ["n"]


def _ledger_target(base_path):
    stem, ext = os.path.splitext(base_path)
    return "%s.%s%s" % (stem, run_id(), ext or ".jsonl")


def _ledger_fh():
    global _LEDGER_FH, _LEDGER_OPENED, _META_WRITTEN
    if _LEDGER_FH is None:
        base = os.environ[ENV_LEDGER]
        path = _ledger_target(base)
        parent = os.path.dirname(os.path.abspath(path))
        if parent:
            os.makedirs(parent, exist_ok=True)
        if os.path.exists(path):
            raise RuntimeError(
                "[StrictFork] refusing to touch existing ledger %r "
                "(unique-ledger-per-run policy; choose a fresh base path)"
                % (path,))
        _LEDGER_FH = open(path, "x", encoding="utf-8", newline="\n")
        _LEDGER_OPENED = path
        _META_WRITTEN = False
    return _LEDGER_FH


def _diffusers_string():
    try:
        import diffusers
        return str(getattr(diffusers, "__version__", "unknown"))
    except Exception:
        return "UNDECLARED"


def _ensure_meta_line():
    global _META_WRITTEN
    if _META_WRITTEN:
        return
    cfg = fork_cfg()
    meta = {
        "event": "meta",
        "pin": PATCH_TARGET_PIN,
        "patch": PATCH_ID,
        "patch_sha256": os.environ.get(ENV_PATCH_SHA) or "UNDECLARED",
        "profile_sha256": os.environ.get(ENV_PROFILE_SHA) or "UNDECLARED",
        "common_config_sha256": os.environ.get(ENV_CONFIG_SHA) or "UNDECLARED",
        "fork_chunk": (int(cfg["fork_chunk"]) if cfg and "fork_chunk" in cfg
                       else None),
        "branch_id": get_branch_id(),
        "run_id": run_id(),
        "pair_id": pair_id(),
        "torch": torch.__version__,
        "diffusers": _diffusers_string(),
        "warp_seed": _warp_seed_record(),
        "continuation": _RESTORED,
        "env_fork_present": os.environ.get(ENV_FORK) is not None,
    }
    _LEDGER_FH.write(json.dumps(meta, sort_keys=True) + "\n")
    _LEDGER_FH.flush()
    _META_WRITTEN = True


def _warp_seed_record():
    value = os.environ.get(ENV_WARP_SEED)
    if value is None:
        return {"present": False, "sha256": None}
    return {"present": True, "sha256": sha256_bytes(value.encode("utf-8"))}


def log_draw(site_id, tensor, generator=None, gen_before=None,
             generator_role="main", extra=None):
    """Record one actual draw. Early-return (zero cost) when the ledger is off."""
    if not ledger_enabled():
        return None
    with _LOCK:
        ordinal = _ORDINALS.get(site_id, 0) + 1
        _ORDINALS[site_id] = ordinal
        entry = _TAPE.record(
            site_id, tensor,
            seq=_next_seq(),
            ordinal=ordinal,
            run_id=run_id(),
            pair_id=pair_id(),
            branch_id=get_branch_id(),
            chunk=_CTX.get("chunk"),
            stage=_CTX.get("stage"),
            generator_role=generator_role,
            generator_state_hash_before=gen_before,
            generator_state_hash_after=gen_state_of(generator),
            global_rng_sha256=global_rng_digest(),
            extra=dict(extra or {}),
        )
        _ledger_fh()
        _ensure_meta_line()
        _LEDGER_FH.write(json.dumps(entry, sort_keys=True) + "\n")
        _LEDGER_FH.flush()
        return entry


def collect_generators(main_generator, geo_patchdrop_gen=None):
    """Names must be stable across branches so sidecars pair up."""
    g = main_generator[0] if isinstance(main_generator, list) else main_generator
    out = {}
    if g is not None:
        out["main"] = g
    if geo_patchdrop_gen is not None:
        out["geo_patchdrop"] = geo_patchdrop_gen
    return out


# ------------------------------------------------------------------ digest --
def _resolve_attr(root, dotted):
    cur = root
    for part in dotted.split("."):
        if isinstance(cur, dict):
            if part not in cur:
                return False, None
            cur = cur[part]
        else:
            if not hasattr(cur, part):
                return False, None
            cur = getattr(cur, part)
    return True, cur


def _sha_of_raw(raw):
    return hashlib.sha256(raw).hexdigest()


def _digest_walk(o):
    """MUST stay byte-compatible with harness.fork_state_digest.digest_value."""
    if o is None:
        return "N:"
    if isinstance(o, torch.Tensor):
        shape_s, dtype_s = str(list(o.shape)), str(o.dtype)
        raw, _, _ = _byte_payload(o)
        pre = shape_s + "|" + dtype_s + "|"
        return "T:%s:%s:%s" % (shape_s, dtype_s, _sha_of_raw(pre.encode() + raw))
    if isinstance(o, torch.Generator):
        st = o.get_state().contiguous().reshape(-1).numpy().tobytes()
        return "GEN:" + _sha_of_raw(st)
    if isinstance(o, bool):
        return "B:" + repr(o)
    if isinstance(o, (int, float, str)):
        return "P:" + repr(o)
    if isinstance(o, dict):
        parts = ";".join("%s=%s" % (repr(k), _digest_walk(o[k]))
                         for k in sorted(o, key=repr))
        return "D{%s}" % parts
    if isinstance(o, (list, tuple)):
        return "L[%s]" % ";".join(_digest_walk(x) for x in o)
    if isinstance(o, (set, frozenset)):
        return "S{%s}" % ";".join(sorted(repr(x) for x in o))
    d = getattr(o, "__dict__", None)
    if d is not None:
        inner = ";".join("%s=%s" % (k, _digest_walk(d[k])) for k in sorted(d))
        return "O<%s>{%s}" % (type(o).__name__, inner)
    return "X:" + repr(o)


def _digest_entry(v):
    return {"status": "OK", "kind": type(v).__name__,
            "sha256": _sha_of_raw(_digest_walk(v).encode("utf-8"))}


def capture_fork_state_digest(pipeline_root, generators=None,
                              fork_chunk=None, branch_id=None):
    """FORK_STATE_DIGEST v1 manifest; mirrors harness.fork_state_digest.capture."""
    fields = {}
    missing_required = []
    for fid, required, paths in STATE_DIGEST_FIELD_SPEC:
        entries = {}
        values = []
        for path in paths:
            ok, v = _resolve_attr(pipeline_root, path)
            if ok:
                e = _digest_entry(v)
                e["path"] = path
                entries[path] = e
                values.append("%s=%s" % (path, e["sha256"]))
            else:
                entries[path] = {"status": "MISSING", "path": path}
        group_sha = _sha_of_raw((";".join(values)).encode("utf-8"))
        present = all(entries[p]["status"] == "OK" for p in paths)
        fields[fid] = {"required": required, "paths": entries,
                       "group_sha256": group_sha, "present": present}
        if required and not present:
            missing_required.extend(pp for pp in paths
                                    if entries[pp]["status"] == "MISSING")
    gens = {}
    for name in sorted(generators or {}):
        g = generators[name]
        if g is None:
            gens[name] = {"status": "MISSING"}
        else:
            st = g.get_state().contiguous().reshape(-1).numpy().tobytes()
            gens[name] = {"status": "OK", "sha256": _sha_of_raw(st),
                          "nbytes": len(st)}
    try:
        gst = torch.get_rng_state().contiguous().reshape(-1).numpy().tobytes()
        global_cpu = _sha_of_raw(gst)
    except Exception:
        global_cpu = None
    fields["F10"] = {"required": True, "generators": gens,
                     "global_cpu_rng_sha256": global_cpu}
    manifest = {
        "schema": DIGEST_SCHEMA,
        "pin": PATCH_TARGET_PIN,
        "branch_id": branch_id if branch_id is not None else get_branch_id(),
        "fork_chunk": fork_chunk,
        "fields": fields,
        "excluded_note": EXCLUDED_NOTE,
    }
    body = json.dumps(manifest, sort_keys=True, separators=(",", ":"))
    manifest["manifest_sha256"] = hashlib.sha256(body.encode("utf-8")).hexdigest()
    if missing_required:
        manifest["missing_required_paths"] = sorted(missing_required)
    return manifest


def digest_canonical(manifest):
    m = dict(manifest)
    m.pop("manifest_sha256", None)
    return json.dumps(m, sort_keys=True, separators=(",", ":"))


def compare_fork_state_digest(parent, child, require_advisory=False):
    """Mirror of harness.fork_state_digest.compare (same report contract)."""
    if isinstance(parent, str):
        parent = json.loads(parent)
    if isinstance(child, str):
        child = json.loads(child)
    mismatches, missing, advisory = [], [], []
    p_ok = parent.get("manifest_sha256") and \
        hashlib.sha256(digest_canonical(parent).encode("utf-8")).hexdigest() \
        == parent["manifest_sha256"]
    c_ok = child.get("manifest_sha256") and \
        hashlib.sha256(digest_canonical(child).encode("utf-8")).hexdigest() \
        == child["manifest_sha256"]
    if not p_ok:
        mismatches.append({"field": "<manifest>", "reason": "PARENT_DIGEST_CORRUPT"})
    if not c_ok:
        mismatches.append({"field": "<manifest>", "reason": "CHILD_DIGEST_CORRUPT"})
    pf, cf = parent.get("fields", {}), child.get("fields", {})
    for fid in sorted(set(pf) | set(cf)):
        pv, cv = pf.get(fid), cf.get(fid)
        if pv is None or cv is None:
            missing.append({"field": fid, "side": "child" if pv else "parent"})
            continue
        pg, cg = pv.get("group_sha256"), cv.get("group_sha256")
        if pg is not None and cg is not None:
            same = (pg == cg)
        else:
            strip = lambda d: json.dumps({k: v for k, v in sorted(d.items())
                                          if k != "required"}, sort_keys=True)
            same = (strip(pv) == strip(cv))
        if not same:
            (advisory if fid in ADVISORY_FIELDS else mismatches).append(
                {"field": fid, "reason": "VALUE_MISMATCH"})
        for side_name, side in (("parent_missing_path", pv),
                                ("child_missing_path", cv)):
            for pp, e in (side.get("paths") or {}).items():
                if e.get("status") == "MISSING":
                    tgt = advisory if fid in ADVISORY_FIELDS else missing
                    tgt.append({"field": fid, side_name: pp})
    return {
        "match": not mismatches and not missing and
                 (not require_advisory or not advisory),
        "mismatches": mismatches,
        "missing": missing,
        "advisory_diffs": advisory,
        "require_advisory": require_advisory,
    }


def _write_event(record):
    if not ledger_enabled():
        return
    with _LOCK:
        _ledger_fh()
        _ensure_meta_line()
        rec = dict(record)
        rec.setdefault("seq", _next_seq())
        rec.setdefault("run_id", run_id())
        rec.setdefault("pair_id", pair_id())
        _LEDGER_FH.write(json.dumps(rec, sort_keys=True) + "\n")
        _LEDGER_FH.flush()


def _patch_meta_continuation():
    """Lazy-meta repair: the meta header is written at first ledger use,
    before restore status is knowable. Once a restore succeeds, rewrite the
    header's continuation field so validator META checks reflect reality."""
    if _LEDGER_FH is None or not getattr(_LEDGER_FH, "name", None):
        return
    path = _LEDGER_OPENED or _LEDGER_FH.name
    _LEDGER_FH.flush()
    try:
        with open(path, "r", encoding="utf-8") as fh:
            lines = fh.readlines()
        if not lines:
            return
        head = json.loads(lines[0])
        if head.get("event") != "meta":
            return
        if head.get("continuation") == CONTINUATION_MARKER:
            return
        head["continuation"] = CONTINUATION_MARKER
        lines[0] = json.dumps(head, sort_keys=True) + "\n"
        _LEDGER_FH.close()
        with open(path, "w", encoding="utf-8", newline="\n") as fh:
            fh.writelines(lines)
        globals()["_LEDGER_FH"] = open(path, "a", encoding="utf-8", newline="\n")
    except (ValueError, OSError):
        pass


def maybe_fork_boundary(chunk_index, generators, pipeline=None):
    """TRUE fork boundary hook: called at the TOP of every chunk iteration;
    acts only at cfg.fork_chunk, before prompt selection / warp render / ANY
    post-fork stochastic draw."""
    cfg = fork_cfg()
    if cfg is None:
        return None
    if int(cfg.get("fork_chunk", -(10 ** 9))) != int(chunk_index):
        return None
    mode = str(cfg.get("mode", "capture")).lower()
    if mode == "capture":
        return _fork_capture(cfg, int(chunk_index), generators, pipeline)
    if mode == "restore":
        return _fork_restore(cfg, int(chunk_index), generators, pipeline)
    raise ValueError("[StrictFork] unknown mode: %r" % (mode,))


def _fork_capture(cfg, chunk_index, generators, pipeline=None):
    out_dir = cfg.get("out_dir")
    if not out_dir:
        raise ValueError("[StrictFork] capture mode requires out_dir")
    if pipeline is None:
        raise ValueError("[StrictFork] capture mode requires the pipeline "
                         "object (FORK_STATE_DIGEST is REQUIRED)")
    os.makedirs(out_dir, exist_ok=True)
    gens_json = {}
    for name in sorted(generators):
        g = generators[name]
        st = g.get_state().cpu().contiguous()
        raw = st.reshape(-1).numpy().tobytes()
        digest = sha256_bytes(raw)
        pt_path = os.path.join(out_dir, "genstate_%s_chunk%d.pt"
                               % (name, chunk_index))
        torch.save({"name": name, "chunk": chunk_index,
                    "pin": PATCH_TARGET_PIN, "state": st}, pt_path)
        gens_json[name] = {"path": pt_path, "sha256": digest, "nbytes": len(raw)}
    sidecar = {
        "kind": "EVOKE_STRICT_FORK_CAPTURE",
        "pin": PATCH_TARGET_PIN,
        "patch": PATCH_ID,
        "branch_id": get_branch_id(),
        "fork_chunk": chunk_index,
        "generators": gens_json,
        "warp_seed": _warp_seed_record(),
        "global_rng_sha256": global_rng_digest(),
        "torch": torch.__version__,
        "created_utc": _utcnow(),
    }
    sc_path = os.path.join(out_dir, "fork_capture_chunk%d.json" % chunk_index)
    with open(sc_path, "w", encoding="utf-8", newline="\n") as fh:
        json.dump(sidecar, fh, indent=2, sort_keys=True)
    # FORK_STATE_DIGEST sidecar next to the generator capture (REQUIRED)
    digest = capture_fork_state_digest(pipeline, generators,
                                       fork_chunk=chunk_index,
                                       branch_id=get_branch_id())
    if digest.get("missing_required_paths"):
        raise RuntimeError(
            "[StrictFork] FORK_STATE_DIGEST missing REQUIRED fields: %s"
            % (sorted(set(digest["missing_required_paths"])),))
    dg_path = os.path.join(out_dir, "fork_state_digest_chunk%d.json"
                           % chunk_index)
    with open(dg_path, "w", encoding="utf-8", newline="\n") as fh:
        fh.write(json.dumps(digest, sort_keys=True,
                            separators=(",", ":")) + "\n")
    print("[StrictFork] FORK_CAPTURE branch=%s chunk=%d gens=%s -> %s "
          "(state digest: %s)"
          % (get_branch_id(), chunk_index, sorted(gens_json), sc_path, dg_path),
          flush=True)
    _write_event({
        "event": "FORK_CAPTURE", "branch_id": get_branch_id(),
        "chunk": chunk_index, "generators": sorted(gens_json),
        "sidecar": sc_path, "state_digest": dg_path,
    })
    return sc_path


def _fork_restore(cfg, chunk_index, generators, pipeline=None):
    side = cfg.get("sidecar")
    if not side:
        raise ValueError("[StrictFork] restore mode requires sidecar")
    if pipeline is None:
        raise ValueError("[StrictFork] restore mode requires the pipeline "
                         "object (FORK_STATE_DIGEST comparison is REQUIRED)")
    with open(side, "r", encoding="utf-8") as fh:
        meta = json.load(fh)
    if meta.get("kind") != "EVOKE_STRICT_FORK_CAPTURE":
        raise RuntimeError("[StrictFork] sidecar kind mismatch: %r"
                           % (meta.get("kind"),))
    if meta.get("pin") != PATCH_TARGET_PIN:
        raise RuntimeError("[StrictFork] sidecar pin %r != patched pin"
                           % (meta.get("pin"),))
    if int(meta.get("fork_chunk", -1)) != chunk_index:
        raise RuntimeError("[StrictFork] sidecar fork_chunk %r != boundary %d"
                           % (meta.get("fork_chunk"), chunk_index))
    restored = []
    for name in sorted(meta.get("generators", {})):
        info = meta["generators"][name]
        g = generators.get(name)
        if g is None:
            raise RuntimeError("[StrictFork] restore: generator %r absent at "
                               "boundary" % (name,))
        obj = torch.load(info["path"], map_location="cpu", weights_only=True)
        st = obj["state"]
        raw = st.contiguous().reshape(-1).numpy().tobytes()
        if sha256_bytes(raw) != info["sha256"]:
            raise RuntimeError("[StrictFork] captured state hash mismatch for %r"
                               % (name,))
        g.set_state(st.to(torch.uint8))
        restored.append(name)
    # ---- FORK_STATE_DIGEST: compare BEFORE any chunk-c* draw -----------------
    parent_path = cfg.get("parent_state_digest")
    if not parent_path:
        raise ValueError("[StrictFork] restore mode requires "
                         "parent_state_digest (boundary-state equivalence is "
                         "REQUIRED under STRICT_NOISE_COUPLED)")
    with open(parent_path, "r", encoding="utf-8") as fh:
        parent_digest = json.load(fh)
    out_dir = os.path.dirname(os.path.abspath(side)) or "."
    child_digest = capture_fork_state_digest(pipeline, generators,
                                             fork_chunk=chunk_index,
                                             branch_id=get_branch_id())
    child_path = os.path.join(out_dir, "fork_state_digest_child_chunk%d.json"
                              % chunk_index)
    with open(child_path, "w", encoding="utf-8", newline="\n") as fh:
        fh.write(json.dumps(child_digest, sort_keys=True,
                            separators=(",", ":")) + "\n")
    report = compare_fork_state_digest(parent_digest, child_digest)
    if not report["match"]:
        detail = {"mismatches": report["mismatches"], "missing": report["missing"]}
        _write_event({
            "event": "FORK_STATE_MISMATCH", "branch_id": get_branch_id(),
            "chunk": chunk_index, "detail": detail,
            "child_state_digest": child_path, "parent_state_digest": parent_path,
        })
        raise RuntimeError("[StrictFork] FORK_STATE_MISMATCH at chunk %d: %s"
                           % (chunk_index, json.dumps(detail, sort_keys=True)))
    global _RESTORED
    with _LOCK:
        _RESTORED = CONTINUATION_MARKER
        _patch_meta_continuation()
        rec = {
            "event": CONTINUATION_MARKER,
            "branch_id": get_branch_id(),
            "chunk": chunk_index,
            "restored_generators": restored,
            "sidecar": side,
            "state_digest_parent": parent_path,
            "state_digest_child": child_path,
        }
    _write_event(rec)
    print("[StrictFork] GENERATOR_STATE_RESTORED branch=%s chunk=%d gens=%s "
          "(state digest MATCH)" % (get_branch_id(), chunk_index, restored),
          flush=True)
    return rec


def continuation_marker():
    return _RESTORED


def _utcnow():
    import datetime
    return datetime.datetime.utcnow().replace(microsecond=0).isoformat() + "Z"
